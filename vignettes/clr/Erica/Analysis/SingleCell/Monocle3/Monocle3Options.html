Monocle3 分析管线配置项。

Declare

            
# namespace Erica.Analysis.SingleCell.Monocle3
export class Monocle3Options {
   # 缓存目录，默认 ./monocle3_cache。
   cacheDir: string;
   # KNN 近邻数，默认 15。
   knnK: integer;
   # 用于 PCA 的高变基因（highly variable genes）数量，默认 2000。 在 PCA 前按表达方差筛选 top N 基因，避免在全基因（数万维）上做 PCA 导致极慢。
   numHVGenes: integer;
   # PCA 主成分数量，默认 50。
   numPCA: integer;
   # 是否强制覆盖缓存（重算所有步骤）。默认 False。
   overwriteCache: boolean;
   # 是否启用计算密集型循环的多核并行化（Parallel.For / PLINQ）。 默认 True。关闭时回退为串行循环，用于验证并行结果与串行逐 bit 一致。
   parallelEnabled: boolean;
   # 是否启用 PseudoVelo 伪速度计算。默认 True。
   pseudoVeloEnabled: boolean;
   # 预留：若改用 LOESS 平滑时的 span 参数（0~1）。默认 0.3。
   pseudoVeloSpan: double;
   # 平滑窗口半宽（实际窗宽 = 2*window+1），默认 2（窗宽 5）。
   pseudoVeloWindow: integer;
   # Louvain 分辨率（传给 Builder.Load 的 eps），默认 1.0。
   resolution: double;
   # 指定的根 cluster（用于伪时间计算）；为 Nothing 时自动选择最外围 cluster。
   rootCluster: Nullable`1;
   # UMAP 嵌入维度，默认 3。
   umapDim: integer;
   # 是否启用缓存（命中则跳过已计算步骤）。默认 True。
   useCache: boolean;
   # 是否使用 Leiden 算法（否则 Louvain）。
   useLeiden: boolean;
   # 是否把细胞伪速度投影到 UMAP2D 坐标（生成 velocityUMAP）。默认 True。
   useVelocityProjection: boolean;
}
